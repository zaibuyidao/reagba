include(FetchContent)
function(reagba_dependency name repo tag)
  if(EXISTS "${CMAKE_SOURCE_DIR}/third_party/${name}/CMakeLists.txt")
    set(FETCHCONTENT_SOURCE_DIR_${name} "${CMAKE_SOURCE_DIR}/third_party/${name}" PARENT_SCOPE)
    string(TOUPPER "${name}" upper)
    set(FETCHCONTENT_SOURCE_DIR_${upper} "${CMAKE_SOURCE_DIR}/third_party/${name}" PARENT_SCOPE)
  endif()
  FetchContent_Declare(${name} GIT_REPOSITORY ${repo} GIT_TAG ${tag} GIT_SHALLOW TRUE)
endfunction()

if(REAGBA_EXTENSION)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/scripts/extension_deps.py" RESULT_VARIABLE deps_status)
  if(NOT deps_status EQUAL 0)
    message(FATAL_ERROR "Could not prepare REAPER SDK/SWELL")
  endif()
  add_library(reaper_reagba SHARED src/extension/Plugin.cpp src/audio/AudioEngine.cpp src/input/InputManager.cpp)
  target_link_libraries(reaper_reagba PRIVATE reagba_core SDL2::SDL2-static)
  target_include_directories(reaper_reagba SYSTEM PRIVATE third_party/reaper-sdk/sdk third_party/WDL/WDL)
  target_compile_definitions(reaper_reagba PRIVATE SDL_MAIN_HANDLED NOMINMAX)
  string(TOLOWER "${CMAKE_SYSTEM_PROCESSOR}" REAGBA_PROCESSOR)
  if(WIN32)
    if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8 OR CMAKE_GENERATOR_PLATFORM MATCHES "[Aa][Rr][Mm]")
      message(FATAL_ERROR "Windows extension requires MSVC x64")
    endif()
    set(REAGBA_ARCH x64)
  elseif(APPLE)
    set_target_properties(reaper_reagba PROPERTIES OBJCXX_STANDARD 17 OBJCXX_STANDARD_REQUIRED YES)
    if(CMAKE_OSX_ARCHITECTURES)
      set(REAGBA_PROCESSOR "${CMAKE_OSX_ARCHITECTURES}")
    endif()
    if(NOT REAGBA_PROCESSOR MATCHES "^(arm64|x86_64)$")
      message(FATAL_ERROR "Build each macOS architecture separately: arm64 or x86_64")
    endif()
    set(REAGBA_ARCH "${REAGBA_PROCESSOR}")
    target_compile_definitions(reaper_reagba PRIVATE SWELL_PROVIDED_BY_APP)
    target_sources(reaper_reagba PRIVATE third_party/WDL/WDL/swell/swell-modstub.mm)
  elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    if(REAGBA_PROCESSOR MATCHES "^(aarch64|arm64)$")
      set(REAGBA_ARCH aarch64)
    elseif(REAGBA_PROCESSOR MATCHES "^(x86_64|amd64)$")
      set(REAGBA_ARCH x86_64)
    else()
      message(FATAL_ERROR "Supported Linux architectures: x86_64, aarch64")
    endif()
    target_compile_definitions(reaper_reagba PRIVATE SWELL_PROVIDED_BY_APP)
    target_sources(reaper_reagba PRIVATE third_party/WDL/WDL/swell/swell-modstub-generic.cpp)
    target_link_libraries(reaper_reagba PRIVATE ${CMAKE_DL_LIBS})
    target_link_options(reaper_reagba PRIVATE "-Wl,--exclude-libs,ALL")
  else()
    message(FATAL_ERROR "Supported extension platforms: Windows, macOS, Linux")
  endif()
  set_target_properties(reaper_reagba PROPERTIES PREFIX "" OUTPUT_NAME "reaper_reagba-${REAGBA_ARCH}")
  install(TARGETS reaper_reagba RUNTIME DESTINATION UserPlugins COMPONENT ReaGBA LIBRARY DESTINATION UserPlugins COMPONENT ReaGBA)
  if(REAGBA_TESTS)

    add_library(reagba_audio_probe SHARED tests/audio_host_probe.cpp)
    target_include_directories(reagba_audio_probe SYSTEM PRIVATE third_party/reaper-sdk/sdk third_party/WDL/WDL)
    target_compile_definitions(reagba_audio_probe PRIVATE NOMINMAX)
    set_target_properties(reagba_audio_probe PROPERTIES PREFIX "" OUTPUT_NAME "reaper_reagba_audio_probe")
    if(WIN32)
      add_test(NAME core_api COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/core_api.py" "$<TARGET_FILE:reaper_reagba>")
    endif()
    add_test(NAME release_packaging COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/ReleaseTests.py")
  endif()
