function(reagba_dependency name repo tag)
  if(EXISTS "${CMAKE_SOURCE_DIR}/third_party/${name}/CMakeLists.txt")
    set(FETCHCONTENT_SOURCE_DIR_${name} "${CMAKE_SOURCE_DIR}/third_party/${name}" PARENT_SCOPE)
    string(TOUPPER "${name}" upper)
    set(FETCHCONTENT_SOURCE_DIR_${upper} "${CMAKE_SOURCE_DIR}/third_party/${name}" PARENT_SCOPE)
  endif()
  FetchContent_Declare(${name} GIT_REPOSITORY ${repo} GIT_TAG ${tag} GIT_SHALLOW TRUE)
endfunction()
